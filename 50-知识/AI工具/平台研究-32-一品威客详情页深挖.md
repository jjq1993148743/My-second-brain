### 抓取方法

```javascript
// 1. 从列表页收集任务 ID
const list = await fetch('https://www.epwk.com/task/');
const ids = [...list.matchAll(/task\.epwk\.com\/(\d+)\//g)].map(m => m[1]);

// 2. 逐个抓详情页
for (const id of ids) {
  const detail = await fetch(`https://task.epwk.com/${id}/`);
  // 提取预算、截稿时间、品类

}
```
